const {spawn}=require('child_process');
const [bin,root]=process.argv.slice(2);
const m=require(root+'/manifest.json');
const rev=m.helixRevision||m.helix_revision, hash=m.runtimeHash||m.runtime_hash, target=m.targetId||m.target_id;
function run(label, initFirst, frames){return new Promise(res=>{
  const c=spawn(bin,['serve','--runtime-root',root]); let buf='',epoch=null,n=0,idx=0;
  console.log(label);
  const send=s=>{console.log('  send  ->',s);c.stdin.write(s+'\n');};
  const next=()=>{ if(idx<frames.length) {send(frames[idx++](epoch)); clearTimeout(t); t=setTimeout(()=>{console.log('  HELPER STILL SERVING 5s after the last frame (no exit) -> killing');c.kill();},5000);} };
  c.stdout.on('data',d=>{buf+=d;let i;while((i=buf.indexOf('\n'))>=0){const l=buf.slice(0,i);buf=buf.slice(i+1);n++;
    const j=JSON.parse(l);
    if(initFirst&&n===1){epoch=j.helperEpoch;console.log('  init  -> ok='+j.ok,'helperEpoch='+epoch);next();}
    else{console.log('  reply ->',l);next();}}});
  c.on('exit',(code)=>{console.log('  EXIT code:',code);clearTimeout(t);res();});
  let t=setTimeout(()=>{console.log('  HELPER STILL SERVING / no exit -> killing');c.kill();},60000);
  if(initFirst) c.stdin.write(JSON.stringify({protocolVersion:1,requestId:1,method:'initialize',params:{expectedHelixRevision:rev,expectedRuntimeHash:hash,target}})+'\n'); else next();
});}
(async()=>{
await run('A: after initialize, {"protocolVersion":2,"id":"a","method":"initialize"}',true,[()=>'{"protocolVersion":2,"id":"a","method":"initialize"}']);
await run('B: after initialize, {"protocolVersion":2,"requestId":"abc","method":"initialize"}',true,[()=>'{"protocolVersion":2,"requestId":"abc","method":"initialize"}']);
await run('C: after initialize, request with helperEpoch = current+1 (then shutdown with the right epoch)',true,[e=>JSON.stringify({protocolVersion:1,requestId:5,helperEpoch:e+1,method:'x'}),e=>JSON.stringify({protocolVersion:1,requestId:6,helperEpoch:e,method:'shutdown'})]);
await run('D: before initialize, request carrying helperEpoch 5 (then shutdown)',false,[()=>JSON.stringify({protocolVersion:1,requestId:3,helperEpoch:5,method:'initialize',params:{}}),()=>JSON.stringify({protocolVersion:1,requestId:4,method:'shutdown'})]);
})();
