const chars = (t) => Array.from(t);
/** char index of the nth occurrence of `needle` in `text`, plus `delta` */
function at(text, needle, delta = 0, nth = 0) {
  const cs = chars(text);
  const ns = chars(needle);
  let seen = -1;
  for (let i = 0; i + ns.length <= cs.length; i++) {
    if (ns.every((c, j) => cs[i + j] === c)) {
      seen++;
      if (seen === nth) return i + delta;
    }
  }
  throw new Error(`needle ${JSON.stringify(needle)} #${nth} not in ${JSON.stringify(text)}`);
}
