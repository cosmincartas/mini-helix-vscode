const MARK = '§';
function mk(lang, text, steps, extra = {}) {
  const chars = Array.from(text);
  const sels = [];
  const clean = [];
  for (const ch of chars) { if (ch === MARK) { sels.push({ anchor: clean.length, head: clean.length }); } else clean.push(ch); }
  const input = { text: clean.join('') };
  if (lang) input.language = lang;
  if (sels.length) input.selections = sels;
  input.steps = steps;
  return { input: Object.assign(input, extra) };
}
